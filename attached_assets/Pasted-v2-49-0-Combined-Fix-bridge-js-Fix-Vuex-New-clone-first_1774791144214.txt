Fix 1 (critical — restores section loading): Exclude Firebase auth URLs from bridge.js

In bridge.js, add this passthrough helper at the top of the file, then apply it to both the XHR and fetch interceptors:

var CF_PASSTHROUGH_URLS = [
  'securetoken.googleapis.com',
  'accounts:signInWithCustomToken',
  'accounts:lookup',
  'identitytoolkit.googleapis.com',
  'firebase:fetch'
];

function cfShouldPassthrough(url) {
  if (!url) return false;
  for (var i = 0; i < CF_PASSTHROUGH_URLS.length; i++) {
    if (url.indexOf(CF_PASSTHROUGH_URLS[i]) !== -1) return true;
  }
  return false;
}

In the XHR open() wrapper, add at the very top:

if (cfShouldPassthrough(url)) {
  return _cf_originalOpen.apply(this, arguments);
}

In the fetch wrapper, add at the very top:

if (cfShouldPassthrough(url)) {
  return _cf_originalFetch.apply(this, arguments);
}

Also wrap every Firestore capture block added in v2.48.x inside try { } catch(e) {} so capture errors can never affect GHL.

Fix 4 (the real fix): New clone-first inject flow in _cf_injectViaBuilderSave

The root cause of all inject failures: the existing page returns 304 Not Modified from browser cache. GHL never re-fetches Firebase Storage for cached pages. A newly created page returns 200 OK (never cached), so GHL fetches Firebase Storage fresh and sections load.

Replace the existing A2 Firebase write flow with this new sequence:

Step 1 — Self-clone the current page:

var clonePayload = {
  funnelId: funnelId,
  funnelIdToImport: funnelId,
  funnels: [funnelId],
  locationId: locationId,
  pageIndexToImport: '0',
  pageIndexToImportInto: '0',
  stepId: stepId,
  stepIdToImportInto: stepId,
  userId: userId
};
var cloneResp = await revex.post('/funnels/clone-funnel-step/', clonePayload);
diag.approach2.cloneStatus = cloneResp.status;
diag.approach2.cloneData = JSON.stringify(cloneResp.data);

Step 2 — Find the new page ID:

var lookupResp = await revex.get('/funnels/lookup/list', {
  params: { funnelId: funnelId, locationId: locationId, type: 'page' }
});
var pages = (lookupResp.data && lookupResp.data.data) ? lookupResp.data.data : [];
var newPage = pages
  .filter(function(p) { return p.stepId === stepId && p._id !== builderId; })
  .sort(function(a, b) { return new Date(b.dateAdded) - new Date(a.dateAdded); })[0];

if (!newPage) {
  diag.approach2.cloneResult = 'ERROR: new page not found after clone';
  // fall through to existing Firebase write as fallback
} else {
  diag.approach2.newPageId = newPage._id;
  diag.approach2.newFirebasePath = newPage.pageDataUrl;
  diag.approach2.newDownloadUrl = newPage.pageDataDownloadUrl;
}

Step 3 — Write AI content to the NEW page's Firebase Storage path:

Use newPage.pageDataUrl as the Firebase Storage write target instead of the existing page's objectPath. The write logic is identical to the existing A2 Firebase PUT — just use newFirebasePath (URL-encoded) as the object path.

After a successful write, run patchToken using newPage.pageDataDownloadUrl as the base URL to restore the original token for the new page.

Step 4 — Navigate the builder to the new page:

var newBuilderUrl = 'https://app.gohighlevel.com/location/' + locationId + '/page-builder/' + newPage._id;
chrome.tabs.update(tab.id, { url: newBuilderUrl });
diag.approach2.navigatedTo = newBuilderUrl;

Step 5 — Update the toast/popup message:

"✓ AI page created and opening in builder now. Sections should load within a few seconds."
